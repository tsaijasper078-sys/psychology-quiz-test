設定後，每次有受測者提交問卷，系統將自動把數據即時推送到您的 Google 試算表中！

超簡單 3 步驟設定 Google 試算表自動同步：
  1. 開啟您的 Google 試算表，點選上方選單的 「擴充功能」 ➔ 「Apps Script」。
  2. 清空裡面的代碼，貼上右側的專用代碼後點擊「儲存」。
  3. 點選右上角 「部署」 ➔ 「新增部署」 ➔ 類型選擇「網頁應用程式」，將「誰可以存取」設為 「所有人 (Anyone)」 點擊部署，並複製產生的網址貼至上方框中即可！
📄 複製下方代碼貼至 Google Apps Script： 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["提交時間", "受測者姓名", "Email/帳號", "第一階段依附類型", "第二階段分類", "第二階段回答細項"]);
  }
  
  var s2Ans = Array.isArray(data.stage2Answers) ? data.stage2Answers.map(function(a, i){ return "Q"+(i+1)+":"+a; }).join(", ") : "";
  
  sheet.appendRow([
    data.submittedAt || new Date().toLocaleString(),
    data.userName || "未命名",
    data.userEmail || "無",
    data.dominantType || "",
    data.stage2Category || "",
    s2Ans
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
